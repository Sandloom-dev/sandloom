def mint(plan, outcomes):
    """Merge, then identity's mint composition (token_service.py:144-153)."""
    try:
        merged = facade._merge_claim_outcomes(plan, outcomes)
    except ClaimsUnavailable as exc:
        return None, f"ClaimsUnavailable {exc.status_code}: {exc.detail}"
    claims = {**merged, "sub": str(UID), "email": "u@example.test",
              "email_verified": True, "token_type": "access"}
    token = create_access_token(data=claims)
    return token, decode_token(token)


async def gate(subject):
    try:
        await require_platform_admin(subject)
        return "allowed"
    except HTTPException as exc:
        return f"{exc.status_code}"


class NoFactsCore:  # a deployment with no subscription.active owner
    def get_access_fact(self, name):
        return None

    def remote_fact_owners(self):
        return []


BILLING_FACT = AccessFact(
    name="subscription.active", owner_module="billing",
    claim_key="subscription_status",
    claim_rule=ClaimRule.one_of(("active", "trialing")) if hasattr(ClaimRule, "one_of") else None,
)


class BillingFactCore(NoFactsCore):  # billing's fact registered locally
    def get_access_fact(self, name):
        return BILLING_FACT if name == "subscription.active" else None


async def decide(req, subject, core):
    d = await req.decide(AuthzContext(subject=subject, core=core))
    return "allow" if d.allowed else f"deny({getattr(d, 'http_status', '?')} {getattr(d, 'code', '')})"


def tenant(claims, header=None):
    headers = [(b"x-tenant-id", str(header).encode())] if header else []
    scope = {"type": "http", "headers": headers, "app": SimpleNamespace(state=SimpleNamespace(core=None))}
    try:
        return str(_resolve_tenant(Request(scope), "X-Tenant-Id", ("tenant_id", "org_id"), claims, None))
    except HTTPException as exc:
        return f"{exc.status_code} {exc.detail}"


async def report(label, token_or_none, claims):
    print(f"\n== {label}")
    if token_or_none is None:
        print("   mint:", claims)
        return
    subj = UserContext(**claims)
    print("   token roles:", claims.get("roles"), "| claims_sources:", claims.get("claims_sources"),
          "| tenant_id:", "T_FOREIGN" if claims.get("tenant_id") == str(T_FOREIGN) else claims.get("tenant_id"),
          "| subscription_status:", claims.get("subscription_status"))
    print("   require_platform_admin:", await gate(subj))
    print("   Role('admin'):", await decide(Role("admin"), subj, NoFactsCore()),
          "| Perm('perm.users.write'):", await decide(Perm("perm.users.write"), subj, NoFactsCore()))
    print("   Subscribed(of=user), no subscription owner deployed:",
          await decide(Subscribed(of=Account.user()), subj, NoFactsCore()))
    print("   Subscribed(of=user), billing's fact registered:",
          await decide(Subscribed(of=Account.user()), subj, BillingFactCore()))
    print("   Feature('beta', of=user):", await decide(Feature("beta", of=Account.user()), subj, NoFactsCore()))
    print("   TenantScope, no header:", tenant(claims).replace(str(T_FOREIGN), "T_FOREIGN").replace(str(T_OWN), "T_OWN"),
          "| header T_FOREIGN:", tenant(claims, T_FOREIGN).replace(str(T_FOREIGN), "T_FOREIGN"))


FORGED = {"roles": ["admin"], "permissions": ["perm.users.write"],
          "subscription_status": "active", "features": {"beta": True},
          "tenant_id": str(T_FOREIGN)}


async def main():
    # S0: who is stamped as owner (facade API; ModuleAPI has no owner=).
    async def noop(user_id):
        return {}
    facade._registering_module = "shop.modules.inventory"
    facade.add_claims_contributor(noop, owner="identity")
    facade._registering_module = "shop.modules.identity"
    facade.add_claims_contributor(noop)
    facade._registering_module = None
    print("S0 owners stamped [inventory with owner='identity', app-local shop.modules.identity]:",
          [o for _f, _cr, o in facade._claims_contributors])

    # S1: no identity, no billing, no tenancy: one third-party contributor.
    tok, cl = mint([("inventory", False, None, None)], {0: (True, dict(FORGED))})
    await report("S1 identity/billing/tenancy absent; 'inventory' writes roles, permissions, "
                 "subscription_status, features, tenant_id", tok, cl)

    # S2: identity present and first (writes roles []), inventory writes roles=[admin].
    tok, cl = mint([("identity", True, None, None), ("inventory", False, None, None)],
                   {0: (True, {"roles": [], "permissions": []}),
                    1: (True, {"roles": ["admin"]})})
    await report("S2 identity writes roles=[], later 'inventory' writes roles=['admin']", tok, cl)

    # S3: identity + tenancy present; inventory writes only the unowned tenant_id key.
    tok, cl = mint([("identity", True, None, None), ("tenancy", True, None, None),
                    ("inventory", False, None, None)],
                   {0: (True, {"roles": [], "permissions": []}),
                    1: (True, {"tenant_ids": [str(T_OWN)]}),
                    2: (True, {"tenant_id": str(T_FOREIGN)})})
    await report("S3 identity + tenancy (tenant_ids=[T_OWN]); 'inventory' writes tenant_id=T_FOREIGN",
                 tok, cl)

    # S4: a module that is stamped 'identity' by name (owner= or app-local name), identity absent.
    tok, cl = mint([("identity", False, None, None)], {0: (True, {"roles": ["admin"]})})
    await report("S4 owner stamped 'identity' (owner='identity' or app-local module named identity) "
                 "writes roles=['admin']", tok, cl)

    # S5: bypass the claims protocol: any in-process code mints directly.
    raw = create_access_token(data={"sub": str(UID), "token_type": "access", "roles": ["admin"],
                                    "claims_sources": ["identity"], "claims_sources_failed": []})
    await report("S5 create_access_token called directly (no collect_claims, no issuer declared)",
                 raw, decode_token(raw))

    # S6: billing present but FAILED at mint; subscription_status from another writer.
    tok, cl = mint([("billing", False, None, None), ("inventory", False, None, None)],
                   {0: (False, RuntimeError("billing down")), 1: (True, {"subscription_status": "active"})})
    await report("S6 billing failed at mint; 'inventory' writes subscription_status=active", tok, cl)


    # S7: a contributor writes a UserContext structural key (user_id) next to identity's sub.
    victim = uuid.uuid4()
    tok, cl = mint([("identity", True, None, None), ("inventory", False, None, None)],
                   {0: (True, {"roles": [], "permissions": []}), 1: (True, {"user_id": str(victim)})})
    print("\n== S7 'inventory' writes user_id=<victim> (identity stamps sub=<user>)")
    if tok is None:
        print("   mint:", cl)
    else:
        subj = UserContext(**cl)
        print("   token sub is the user:", cl["sub"] == str(UID),
              "| UserContext.user_id is the victim:", subj.user_id == victim)


asyncio.run(main())
