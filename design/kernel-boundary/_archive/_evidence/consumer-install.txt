async def can_order(org_id: UUID, user_id: UUID) -> bool:
    client = membership.client()
    if client is not None and not await client.is_member(org_id, user_id):

"""Consumer install probe: sandloom + sandloom-billing wheels, tenancy absent.

Phase A (no binding): bootstrap, contract registration, optional consumer.
Phase B (--rpc): a stand-in provider of tenancy.membership.v1 (NOT the tenancy
module: a plain class served by the kernel's RPC routes) on a uvicorn port; the
billing process binds it through SERVICE_BINDINGS and calls it, results
converted by the kernel wheel's DTO copy (sandloom.shared.contracts.tenancy_dto).
"""
import asyncio
import json
import os
import socket
import sys
import uuid

RPC = "--rpc" in sys.argv
os.environ.update({
    "ENVIRONMENT": "dev",
    "DATABASE_URL": "postgresql+asyncpg://u:p@localhost/db",
    "CACHE_URL": "redis://localhost:6379/0",
    "BROKER_URL": "redis://localhost:6379/1",
    "JWT_SECRET": "consumer-probe-not-a-real-secret",
    "EVENT_BUS_BACKEND": "memory",
})
os.environ.pop("SANDLOOM_MODULES", None)  # default: installed entry points
PORT = 0
if RPC:
    from sandloom.core.signing_keys import (
        generate_private_key, generate_x25519_key, private_pem, public_pem,
        x25519_private_pem, x25519_public_pem,
    )
    sig, dh = generate_private_key("EdDSA"), generate_x25519_key()
    with socket.socket() as s:
        s.bind(("127.0.0.1", 0))
        PORT = s.getsockname()[1]
    sid = "probe"
    os.environ.update({
        "RPC_SERVICE_ID": sid, "RPC_PRIVATE_KEY": private_pem(sig), "RPC_KEY_ID": "k1",
        "RPC_DH_PRIVATE_KEY": x25519_private_pem(dh), "RPC_DH_KEY_ID": "k1",
        "RPC_SERVICE_KEYS": json.dumps({sid: {"k1": public_pem(sig)}}),
        "RPC_SERVICE_DH_KEYS": json.dumps({sid: {"k1": x25519_public_pem(dh)}}),
        "RPC_ALLOWED_CALLS": json.dumps({sid: ["tenancy.membership.v1"]}),
        "SERVICE_BINDINGS": json.dumps({"tenancy.membership.v1": {
            "base_url": f"http://127.0.0.1:{PORT}", "service_id": sid}}),

if RPC:
    import uvicorn
    from fastapi import FastAPI

    from sandloom.core.facade import CoreFacade
    from sandloom.core.module_api import contract_method
    from sandloom.core.rpc import close_rpc_client, install_service_rpc_routes

    class _UoW:
        session = None

        async def __aenter__(self):
            self.session = type("S", (), {"info": {}})()
            return self

        async def __aexit__(self, *exc):
            self.session = None

    class _Database:
        def uow_factory(self):
            return _UoW

        def outbox(self):
            return None

        def audit(self):
            return None

    class StandInMembership:  # stands in for tenancy's provider, no tenancy code
        @classmethod
        @contract_method
        async def get_member_state(cls, organization_id: uuid.UUID, user_id: uuid.UUID) -> dict:
            if organization_id == uuid.UUID(int=0):  # an older wire shape
                return {"role": "admin", "version": 7}
            return {"organization_id": str(organization_id), "user_id": str(user_id),
                    "role": "admin", "version": 7}

        @classmethod
        @contract_method
        async def has_role_at_least(cls, organization_id: uuid.UUID, user_id: uuid.UUID, minimum: str) -> bool:
            return minimum in ("member", "admin")

    papp = FastAPI()
    pcore = CoreFacade(app=papp)
    papp.state.core = pcore
    pcore.database = lambda name="default": _Database()
    pcore.register_service("tenancy.membership.v1", StandInMembership)
    install_service_rpc_routes(papp, pcore)

    async def main():
        server = uvicorn.Server(uvicorn.Config(papp, host="127.0.0.1", port=PORT,
                                               log_level="warning", lifespan="off"))
        task = asyncio.create_task(server.serve())
        while not server.started:
            await asyncio.sleep(0.01)
        try:
            c = membership.client()
            show("bound client", type(c).__name__)
            st = await c.get_member_state(uuid.uuid4(), uuid.uuid4())
            show("get_member_state over RPC -> type", f"{type(st).__module__}.{type(st).__name__}")
            show("get_member_state over RPC -> value", st)
            show("isinstance(result, kernel-wheel MemberState)", isinstance(st, MemberState))
            try:
                await c.get_member_state(uuid.UUID(int=0), uuid.uuid4())
                show("get_member_state, provider omits ids", "accepted")
            except Exception as exc:  # noqa: BLE001
                show("get_member_state, provider omits ids", f"{type(exc).__name__}: {str(exc).splitlines()[0]}")
            show("has_role_at_least(admin) over RPC", await c.has_role_at_least(uuid.uuid4(), uuid.uuid4(), "admin"))
            show("has_role_at_least(owner) over RPC", await c.has_role_at_least(uuid.uuid4(), uuid.uuid4(), "owner"))
            from sandloom.shared.authz.live_contracts import get_live_adapter
            show("live adapter 'membership' -> contract", get_live_adapter("membership").service)
        finally:
            await close_rpc_client()
            server.should_exit = True
            await task

    asyncio.run(main())
