module installed), composed with SANDLOOM_MODULES = argv[1] (a JSON list of
app-local stand-in modules, or []). Real create_app, real lifespan, real
collect_claims through the contract runtime, real platform_operator gate.
Postgres and Redis are throwaway containers (SANDLOOM_GMAP_PG / _REDIS)."""
import asyncio
import json
import os
import sys
import uuid

HERE = os.path.dirname(os.path.abspath(__file__))
sys.path.insert(0, os.path.join(os.path.dirname(HERE), "app"))
modules = sys.argv[1] if len(sys.argv) > 1 else "[]"
os.environ.update({
    "DATABASE_URL": os.environ["SANDLOOM_GMAP_PG"],
    "CACHE_URL": os.environ["SANDLOOM_GMAP_REDIS"] + "/5",
    "BROKER_URL": os.environ["SANDLOOM_GMAP_REDIS"] + "/6",
    "JWT_SECRET": "gmap-probe-not-a-real-secret-0123456789abcdef",
    "EVENT_BUS_BACKEND": "memory", "SANDLOOM_MODULES": modules,
    "OUTBOX_RELAY_ENABLED": "off", "ENVIRONMENT": "dev",
})
from fastapi.testclient import TestClient  # noqa: E402

from sandloom.core.app_factory import create_app  # noqa: E402
from sandloom.core.auth.tokens import create_access_token  # noqa: E402

app = create_app()
core = app.state.core
print("SANDLOOM_MODULES =", modules)
print("loaded:", list(core.loaded_modules))
print("claims contributors (owner, critical):",
      [(o, c) for _f, c, o in core._claims_contributors])
print("services -> owner:",
      {n: core.contract_owner(n) for n in sorted(core.service_registry.list())})
print("consumers of core.auth.*:",
      [(getattr(d, "channel", None), getattr(d, "name", None)) for d in core.event_consumers
       if str(getattr(d, "channel", "")).startswith("core.auth.")])
prefixes = sorted({r.path.split("/")[4] for r in app.routes
                   if getattr(r, "path", "").startswith("/api/v1/platform/")})
print("platform routers mounted:", prefixes)

with TestClient(app) as client:
    rr = client.get("/ready"); print("lifespan: started; /ready ->", rr.status_code, rr.text[:300])
    uid = uuid.uuid4()

    async def collect():
        return await core.collect_claims(uid)

    try:
        claims = client.portal.call(collect)
        print("collect_claims ->", json.dumps({k: claims[k] for k in sorted(claims)}))
    except Exception as exc:  # noqa: BLE001
        claims = None
        print("collect_claims raised", type(exc).__name__, getattr(exc, "detail", exc))
    if claims is not None:
        token = create_access_token(data={**claims, "sub": str(uid), "token_type": "access"})
        for prefix in prefixes:
            r = client.get(f"/api/v1/platform/{prefix}/services",
                           headers={"Authorization": f"Bearer {token}"})
            print(f"GET /api/v1/platform/{prefix}/services ->", r.status_code, r.text[:90])
    if "shop.modules.orgs" in core.loaded_modules:
        from sandloom.shared.authz import OrgRole
        from sandloom.shared.authz.context import AuthzContext
        from sandloom.shared.subject import UserContext

        async def org_role():
            subj = UserContext(user_id=uid, claims_sources=[], claims_sources_failed=[])
            out = {}
            for minimum in ("member", "admin"):
                req = OrgRole(minimum, org_param="org")
                ctx = AuthzContext(subject=subj, core=core)
                ctx._org_override = None
                from starlette.requests import Request
                scope = {"type": "http", "headers": [], "path_params": {"org": str(uuid.uuid4())},
                         "app": app, "query_string": b""}
                d = await req.decide(AuthzContext(subject=subj, request=Request(scope), core=core))
                out[minimum] = "allow" if d.allowed else f"deny {d.http_status}"
            return out

        print("OrgRole via the kernel's membership adapter (tenancy.membership.v1 served by orgs):",
              client.portal.call(org_role))

OrgRole live adapter calls)."""
from uuid import UUID

from sandloom.sdk.contracts import contract


@contract("tenancy.membership.v1")
class OrgsMembership:
    async def has_role_at_least(self, org_id: UUID, user_id: UUID, minimum: str) -> bool:
        return minimum == "member"
